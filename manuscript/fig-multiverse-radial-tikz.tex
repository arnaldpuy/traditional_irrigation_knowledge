% Radial version of the forking-paths figure. The 475 assumptions sit at the
% centre; the five choices of the multiverse analysis (SM Table "design") are
% concentric rings, each with its true number of options (community 8,
% knowledge domain 5, prompt 3, LLM family 3, vote rule 3), so the half-fan
% ends in exactly 8 x 5 x 3 x 3 x 3 = 1,080 leaves. Around the rim, one radial
% bar per leaf shows the irreducible (red) and reducible (blue) shares of that
% specification, from 0% (inner edge) to 100% (outer edge), read from
% fig-multiverse-shares.tex (code/79_fig_multiverse_shares.R). a) A conventional
% analysis follows one path, the reference specification. b) The multiverse
% analysis follows every path.

\definecolor{irr}{HTML}{b2182b}
\definecolor{leg}{HTML}{2166ac}
\colorlet{guide}{black!9}
% 35% tints on white, as the alpha = 0.35 violins of the shares figure
\colorlet{irrsoft}{irr!35}
\colorlet{legsoft}{leg!35}

\newcommand{\rzero}{0.30}   % radius of the root arc (cm)
\newcommand{\dr}{0.52}      % spacing between rings (cm)
\newcommand{\rin}{3.02}     % inner edge of the share bars (cm), 0%
\newcommand{\rout}{3.72}    % outer edge of the share bars (cm), 100%
\newcommand{\leaf}{\refleaf}

% Levels: level / nodes / leaves under each node / options / line width (pt)
\newcommand{\levels}{1/8/135/8/0.9, 2/40/27/5/0.6, 3/120/9/3/0.4, 4/360/3/3/0.22, 5/1080/1/3/0.1}

% angle (deg) of node j at a level whose nodes hold \span leaves each
\newcommand{\ang}[2]{180 - ((#1) + 0.5) * (#2) / 6}
\newcommand{\rad}[1]{\rzero + (#1) * \dr}

% Whole tree: arcs at the parent ring, radial spokes out to the child ring
\newcommand{\radialtree}[1]{%
  \draw[#1, line width=0.9pt] (0, 0) -- (90:\rzero);
  \foreach \l/\nl/\span/\b/\lw in \levels {
    \pgfmathtruncatemacro{\np}{\nl / \b - 1}
    \pgfmathsetmacro{\rp}{\rad{\l - 1}}
    \pgfmathsetmacro{\rc}{\rad{\l}}
    \foreach \p in {0,...,\np} {
      \pgfmathsetmacro{\ta}{\ang{\p * \b}{\span}}
      \pgfmathsetmacro{\tb}{\ang{\p * \b + \b - 1}{\span}}
      \draw[#1, line width=\lw pt] (\ta:\rp) arc[start angle=\ta, end angle=\tb, radius=\rp];
    }
    \pgfmathtruncatemacro{\last}{\nl - 1}
    \foreach \j in {0,...,\last} {
      \pgfmathsetmacro{\t}{\ang{\j}{\span}}
      \draw[#1, line width=\lw pt] (\t:\rp) -- (\t:\rc);
    }
  }
}

% The single path to \leaf
\newcommand{\onepath}[1]{%
  \draw[#1] (0, 0) -- (90:\rzero);
  \foreach \l/\nl/\span/\b/\lw in \levels {
    \pgfmathtruncatemacro{\onj}{floor(\leaf / \span)}
    \pgfmathtruncatemacro{\onp}{floor(\leaf / (\span * \b))}
    \pgfmathsetmacro{\tc}{\ang{\onj}{\span}}
    \pgfmathsetmacro{\tp}{\ang{\onp}{\span * \b}}
    \pgfmathsetmacro{\rp}{\rad{\l - 1}}
    \pgfmathsetmacro{\rc}{\rad{\l}}
    \draw[#1] (\tp:\rp) arc[start angle=\tp, end angle=\tc, radius=\rp] -- (\tc:\rc);
    \fill[black] (\tc:\rc) circle (1.25pt);
  }
}

\newcommand{\rings}{%
  \foreach \l in {1,...,5} {
    \pgfmathsetmacro{\r}{\rad{\l}}
    \draw[guide, line width=0.35pt] (180:\r) arc[start angle=180, end angle=0, radius=\r];
  }
  \draw[guide, line width=0.35pt] (-\rout - 0.1, 0) -- (\rout + 0.1, 0);
}

\newcommand{\rootnode}{%
  \fill[black] (0, 0) circle (1.7pt);
  \node[font=\scriptsize] at (0, -0.55) {475 assumptions};
}
